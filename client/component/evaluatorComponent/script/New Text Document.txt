export const HeaderTable=()=>{

    const label=[]

    label.push({

        text:'Date',

        className:'dateListEval'

    })

    label.push({

        text:'Author',

        className:'authorListEval'

    })

    label.push({

        text:'Campus',

        className:'campusListEval'

    })

    const gethead=(header)=>{

        label.forEach(val=>{

            header.appendChild($({

                tag:'td',

                text:val.text,

                att:{

                    className:val.className

                }

            }))

        })

    }

    return($({

        tag:'table',

        att:{

            className:'headerList'

        },

        child:[

            $({

                tag:'tr',

                elementHandler:gethead

            })

        ]

    }))

}
